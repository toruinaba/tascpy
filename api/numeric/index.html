    Numeric モジュール

    tascpy.numeric の API リファレンスです。numpy の配列を受け取って配列・値を返す解析関数で、tascpy のコレクションを使わずに単体で使えます。

    import numpy as np
from tascpy.numeric.hysteresis import compute_cumulative_curve, compute_skeleton_curve

disp = np.array([0.0, 1.0, 0.0, -1.0, 0.0, 2.0, 0.0])
load = np.array([0.0, 10.0, 0.0, -10.0, 0.0, 15.0, 0.0])

# サイクル番号（markers）は省略すると荷重の符号反転から数える
d_ske, p_ske = compute_skeleton_curve(disp, load)
d_cum, p_cum = compute_cumulative_curve(disp, load)

    引数は常に (x, y) の順です。汎用の x-y 曲線（curve）は x, y、繰り返し載荷（hysteresis）は displacement, load と書きます。

    def diff(
    x: Union[np.ndarray, List[float]],
    y: Union[np.ndarray, List[float]],
    method: str = "central"
) -> np.ndarray:
    """x に対する y の微分係数 dy/dx を計算します。

    差分を計算できない点は NaN です（x が変化しない区間、forward の最後の点、backward の最初の点）。
    端の点を隣の値で埋めることはしません。

    Args:
        x (Union[np.ndarray, List[float]]): x座標の配列。
        y (Union[np.ndarray, List[float]]): y座標の配列。
        method (str, optional): 微分方法 ('central', 'forward', 'backward')。デフォルトは "central"。

    Returns:
        np.ndarray: 計算された微分係数の配列。

    Raises:
        ValueError: xとyの長さが異なる場合、またはデータ点が2点未満の場合、または無効なメソッドが指定された場合。

    Examples:
        >>> import numpy as np
        >>> from tascpy.numeric.math import diff
        >>> x, y = np.array([0.0, 1.0, 1.0, 2.0, 4.0]), np.array([0.0, 10.0, 12.0, 20.0, 30.0])
        >>> diff(x, y, method="backward")
        array([nan, 10., nan,  8.,  5.])
        >>> diff(x, y, method="forward")
        array([10., nan,  8.,  5., nan])
    """
    y_arr = np.asanyarray(y, dtype=float)
    x_arr = np.asanyarray(x, dtype=float)

    if len(x_arr) != len(y_arr):
        raise ValueError("Length of x and y must be same")
    if len(x_arr) < 2:
        raise ValueError(f"有効なデータポイントが不足しています: {len(x_arr)} (最低2点必要)")
    if method not in ("central", "forward", "backward"):
        raise ValueError("Invalid method. Use 'central', 'forward', or 'backward'")

    with np.errstate(divide="ignore", invalid="ignore"):
        if method == "central":
            result = np.gradient(y_arr, x_arr)
        else:
            slopes = np.diff(y_arr) / np.diff(x_arr)
            result = np.append(slopes, np.nan) if method == "forward" else np.insert(slopes, 0, np.nan)
    return np.where(np.isfinite(result), result, np.nan)

    def integrate(
    x: Union[np.ndarray, List[float]],
    y: Union[np.ndarray, List[float]],
    method: str = "trapezoid",
    initial_value: float = 0.0
) -> np.ndarray:
    """x に対する y の累積積分 ∫y dx を、データの並び順に沿って台形則で計算します。

    x が増減する場合（荷重-変位の往復など）も並べ替えず、経路に沿って積分します。
    先頭の値は initial_value です。

    Args:
        x (Union[np.ndarray, List[float]]): x の配列。
        y (Union[np.ndarray, List[float]]): y の配列。
        method (str, optional): 積分方法。"trapezoid"（台形則）のみ。
        initial_value (float, optional): 積分の初期値。デフォルトは 0.0。

    Returns:
        np.ndarray: 累積積分の配列（欠損を含む区間以降は NaN）。

    Raises:
        ValueError: サポートされていないメソッドの場合、x と y の長さが異なる場合、2点未満の場合。

    Examples:
        >>> import numpy as np
        >>> from tascpy.numeric.math import integrate
        >>> integrate([10.0, 11.0, 12.0], [1.0, 1.0, 1.0])
        array([0., 1., 2.])
        >>> integrate([0.0, 1.0, 0.0], [0.0, 2.0, 2.0])
        array([ 0.,  1., -1.])
    """
    if method != "trapezoid":
        raise ValueError("現在は trapezoid 積分のみサポートしています")

    y_arr = np.asarray(y, dtype=float)
    x_arr = np.asarray(x, dtype=float)
    if len(x_arr) != len(y_arr):
        raise ValueError("x と y の長さが一致しません")
    if len(x_arr) < 2:
        raise ValueError(f"有効なデータポイントが不足しています: {len(x_arr)} (最低2点必要)")

    areas = np.diff(x_arr) * (y_arr[1:] + y_arr[:-1]) / 2.0
    return initial_value + np.concatenate([[0.0], np.cumsum(areas)])

    def log(values: np.ndarray, base: float = math.e) -> np.ndarray:
    """対数(log)を計算します。

    0以下の値はNaNになります。

    Args:
        values (np.ndarray): 入力値の配列。
        base (float, optional): 対数の底。デフォルトは e（自然対数）。

    Returns:
        np.ndarray: 計算結果の配列。
    """
    # Ensure values are float
    values = values.astype(float)

    with np.errstate(divide='ignore', invalid='ignore'):
        if base == math.e:
            res_arr = np.log(values)
        elif base == 10:
            res_arr = np.log10(values)
        else:
            res_arr = np.log(values) / np.log(base)

    # Mask <=0 to NaN (consistent with original implementation)
    # Note: np.log(-1) is NaN, but np.log(0) is -inf. 
    # Original explicitly set <=0 to NaN.
    mask_le_zero = (values <= 0)
    res_arr[mask_le_zero] = np.nan

    return res_arr

    def normalize(values: np.ndarray, method: str = "minmax") -> np.ndarray:
    """値を正規化します。

    Args:
        values (np.ndarray): 入力値の配列。
        method (str, optional): 正規化方法。
            'minmax': 最小値を0、最大値を1にスケーリング。
            'zscore': 平均を0、標準偏差を1に標準化。デフォルトは "minmax"。

    Returns:
        np.ndarray: 正規化された配列。

    Raises:
        ValueError: 指定されたメソッドが無効な場合。

    Examples:
        >>> from tascpy.numeric.math import normalize
        >>> import numpy as np
        >>> normalize(np.array([0, 5, 10]), method="minmax")
        array([0. , 0.5, 1. ])
    """
    values = values.astype(float)
    result_arr = np.full(len(values), np.nan)

    valid_mask = ~np.isnan(values)
    if not np.any(valid_mask):
         return result_arr

    valid_arr = values[valid_mask]

    if method == "minmax":
        min_val = np.min(valid_arr)
        max_val = np.max(valid_arr)

        if max_val == min_val:
            result_arr[valid_mask] = 0.5
        else:
            # We want result corresponding to original shape
            # values[valid_mask] -> normalized
            norm_vals = (valid_arr - min_val) / (max_val - min_val)
            result_arr[valid_mask] = norm_vals

    elif method == "zscore":
        mean = np.mean(valid_arr)
        variance = np.var(valid_arr)

        if variance < 1e-10:
             result_arr[valid_mask] = 0.0
        else:
             std_dev = np.sqrt(variance)
             norm_vals = (valid_arr - mean) / std_dev
             result_arr[valid_mask] = norm_vals

    else:
        raise ValueError(f"methodは['minmax', 'zscore']のいずれかを指定してください")

    return result_arr

    def detect_outliers(
    values: Any,
    window_size: int = 3,
    threshold: float = 0.5,
    min_abs_value: float = 1e-10,
    scale_factor: float = 1.0,
) -> np.ndarray:
    """移動平均からの偏差率が閾値を超える点を外れ値とみなします。

    偏差率 = |値 - 移動平均| / max(|移動平均|, 標準偏差 × scale_factor, min_abs_value)

    Args:
        values: 入力値の配列（None は欠損）
        window_size: 移動平均のウィンドウサイズ
        threshold: 外れ値とみなす偏差率の閾値
        min_abs_value: 偏差の最小値（これ以下の偏差は外れ値としない、ゼロ除算防止にも使う）
        scale_factor: 標準偏差にかける係数

    Returns:
        np.ndarray: 外れ値フラグの配列（0: 正常, 1: 外れ値、欠損は 0）

    Raises:
        ValueError: ウィンドウサイズが不正な場合、または有効なデータがない場合
    """
    if window_size < 1:
        raise ValueError("ウィンドウサイズは1以上である必要があります")
    values = _as_float(values)
    if len(values) < window_size:
        raise ValueError("データ長がウィンドウサイズより小さいです")
    valid = values[~np.isnan(values)]
    if len(valid) == 0:
        raise ValueError("有効なデータがありません")

    reference = max(float(np.std(valid)) * scale_factor, min_abs_value)
    average = moving_average(values, window_size=window_size)
    deviation = np.abs(values - average)
    with np.errstate(invalid="ignore"):
        ratio = deviation / np.maximum(np.abs(average), reference)
        return np.where((ratio > threshold) & (deviation > min_abs_value), 1, 0)

    def gaussian_filter(
    values: Any,
    sigma: float = 1.0,
    window_size: Optional[int] = None,
) -> np.ndarray:
    """ガウシアンフィルタを適用して平滑化を行います。

    欠損値は無視して畳み込み計算を行います。

    Args:
        values (Union[np.ndarray, List[float]]): 入力値の配列またはリスト。
        sigma (float, optional): ガウス分布の標準偏差。デフォルトは 1.0。
        window_size (Optional[int], optional): フィルタのウィンドウサイズ。指定しない場合は sigma から自動計算されます。

    Returns:
        np.ndarray: 平滑化後の配列。

    Raises:
        ValueError: ウィンドウサイズが1未満の場合。

    Examples:
        >>> from tascpy.numeric.stats import gaussian_filter
        >>> import numpy as np
        >>> arr = np.array([0, 10, 0, 0, 0])
        >>> smoothed = gaussian_filter(arr, sigma=1.0)
    """
    if window_size is None:
        window_size = 2 * int(4.0 * sigma + 0.5) + 1
    if window_size < 1:
        raise ValueError("ウィンドウサイズは1以上である必要があります")
    values = _as_float(values)

    radius = window_size // 2
    x = np.arange(-radius, radius + 1)
    kernel = np.exp(-(x**2) / (2 * sigma**2))
    kernel /= kernel.sum()

    valid = ~np.isnan(values)

    def _blend_core(
    v1: np.ndarray,
    v2: np.ndarray,
    comp_arr: np.ndarray,
    start: float,
    end: float,
    blend_func: Callable[[np.ndarray], np.ndarray],
) -> np.ndarray:
    """内部用: 指定範囲で配列をブレンドします。

    Args:
        v1 (np.ndarray): ブレンド開始前の値の配列。
        v2 (np.ndarray): ブレンド終了後の値の配列。
        comp_arr (np.ndarray): 比較対象の配列。
        start (float): ブレンド開始値。
        end (float): ブレンド終了値。
        blend_func (Callable[[np.ndarray], np.ndarray]): [0, 1] 区間の補間関数。

    Returns:
        np.ndarray: ブレンド後の配列。
    """
    v1 = np.asarray(v1)
    v2 = np.asarray(v2)
    comp_arr = np.asarray(comp_arr)

    # Create masks
    mask_before = comp_arr < start
    mask_after = comp_arr > end
    mask_blend = ~mask_before & ~mask_after

    result = np.empty_like(v1, dtype=float) # Force float for blending

    # Before and After
    result[mask_before] = v1[mask_before]
    result[mask_after] = v2[mask_after]

    # Blend region
    if np.any(mask_blend):
        vals_blend = comp_arr[mask_blend]
        t = (vals_blend - start) / (end - start)

        # Apply blend func (vectorized)
        t_trans = blend_func(t)

        b1 = v1[mask_blend]
        b2 = v2[mask_blend]
        result[mask_blend] = b1 * (1 - t_trans) + b2 * t_trans

    return result

    def _yield_point_general(
    x: np.ndarray, y: np.ndarray, initial_slope: float, factor: float

    """一般降伏法による降伏点"""

    if not np.any(unique_mask) or np.sum(unique_mask) < 2:
        raise ValueError(
            "x がすべて同じか、勾配計算に十分な変動がありません。"
            "事前に .ops.remove_consecutive_duplicates_across() を実行してください。"

    unique_y = y[unique_mask]

        "factor": factor,
        "threshold": float(threshold),
        "slopes_stats": {
            "min": float(np.min(slopes)),
            "max": float(np.max(slopes)),
            "mean": float(np.mean(slopes)),
        },
    }

    if len(yield_idx) > 0:
        idx = yield_idx[0]
        return True, float(unique_x[idx]), float(unique_y[idx]), debug_info

    return False, np.nan, np.nan, debug_info

    def _yield_point_offset(
    x: np.ndarray, y: np.ndarray, initial_slope: float, offset_value: float

    """オフセット法による降伏点（オフセットの向きは最大値の符号に合わせる）"""
    offset_amount = initial_slope * offset_value * np.sign(_peak(y))
    offset_line = initial_slope * x - offset_amount
    diff = y - offset_line

    debug_info = {
        "offset_value": offset_value,
        "offset_amount": float(offset_amount),
        "diff_stats": {
            "min": float(np.min(diff)),
            "max": float(np.max(diff)),
            "mean": float(np.mean(diff)),
            "has_sign_change": bool(any(diff[i - 1] * diff[i] <= 0 for i in range(1, len(diff)))),
        },
    }

    for i in range(1, len(diff)):
        if diff[i - 1] * diff[i] <= 0:
            ratio = abs(diff[i - 1]) / (abs(diff[i - 1]) + abs(diff[i]))
            yield_x = x[i - 1] + ratio * (x[i] - x[i - 1])
            yield_y = y[i - 1] + ratio * (y[i] - y[i - 1])
            return True, float(yield_x), float(yield_y), debug_info

    return False, np.nan, np.nan, debug_info

    def compute_peaks_and_valleys(
    values: np.ndarray,
    distance: int = 1,
    threshold: Optional[float] = None,
) -> np.ndarray:
    """配列内の極大値(1)と極小値(-1)を検出します。

    Args:
        values: 対象の配列（欠損は NaN）
        distance: 前後 distance 点の中で最大（最小）である点だけを残す
        threshold: 隣接点との差がこれ未満のピーク（バレー）は除く

    Returns:
        np.ndarray: ピーク(1)、バレー(-1)、その他(0)のフラグ配列

    Examples:
        >>> from tascpy.numeric.curve import compute_peaks_and_valleys
        >>> import numpy as np
        >>> compute_peaks_and_valleys(np.array([0, 2, 1, 0, -1, -2, -1, 0]))
        array([ 0,  1,  0,  0,  0, -1,  0,  0])
    """
    v = np.asarray(values, dtype=float)
    n = len(v)
    flags = np.zeros(n, dtype=int)
    dist = max(1, distance)

    for i in range(1, n - 1):
        if np.isnan(v[i]):
            continue

        if v[i] > v[i - 1] and v[i] > v[i + 1]:
            is_peak = not np.any(v[max(0, i - dist):i] >= v[i])
            if is_peak:
                is_peak = not np.any(v[i + 1:min(n, i + dist + 1)] >= v[i])
            if is_peak and threshold is not None:
                if v[i] - min(v[max(0, i - 1)], v[min(n - 1, i + 1)]) < threshold:
                    is_peak = False
            if is_peak:
                flags[i] = 1

        if v[i] < v[i - 1] and v[i] < v[i + 1]:
            is_valley = not np.any(v[max(0, i - dist):i] <= v[i])
            if is_valley:
                is_valley = not np.any(v[i + 1:min(n, i + dist + 1)] <= v[i])
            if is_valley and threshold is not None:
                if max(v[max(0, i - 1)], v[min(n - 1, i + 1)]) - v[i] < threshold:
                    is_valley = False
            if is_valley:
                flags[i] = -1

    return flags

        range_end: 最大値に対する範囲の終了の割合
        method: "linear_regression"（最小二乗の傾き）または "secant"（範囲の両端を結ぶ傾き）

        float: 剛性

        >>> from tascpy.numeric.curve import compute_stiffness
        >>> import numpy as np
        >>> compute_stiffness(np.array([0, 1, 2, 3, 4]), np.array([0, 100, 200, 300, 400]))
        100.0
    """

    x, y = x[valid], y[valid]

        raise ValueError("剛性計算に十分なデータがありません")

    in_range = (y >= lower) & (y <= upper)
    range_x, range_y = x[in_range], y[in_range]

    def _cycle_groups(displacement: np.ndarray, load: np.ndarray,
                  markers: np.ndarray) -> Tuple[np.ndarray, List[Tuple[List[float], List[float], List[int]]]]:
    """サイクル番号（昇順）と、サイクルごとの (荷重, 変位, 元の行 index) リスト。サイクル番号が欠損の行は除く"""
    load = np.asarray(load, dtype=float)
    displacement = np.asarray(displacement, dtype=float)
    markers = np.asarray(markers, dtype=float)
    keys = np.unique(markers[~np.isnan(markers)])
    groups = []
    for k in keys:
        rows = np.flatnonzero(markers == k)
        groups.append((load[rows].tolist(), displacement[rows].tolist(), rows.tolist()))
    return keys, groups

    def _extend_edge(x: List[float], y: List[float], target: float, target_type: str = "x",
                 position: str = "end") -> Tuple[float, float]:
    """端の2点を結ぶ直線を、x または y が target になる点まで延長した (x, y)（1点しかなければ水平・垂直に延長）"""
    if position == "end":
        if len(x) < 2 or len(y) < 2:
            return (target, y[-1]) if target_type == "x" else (x[-1], target)
        x1, y1, x2, y2 = x[-2], y[-2], x[-1], y[-1]
    else:
        if len(x) < 2 or len(y) < 2:
            return (target, y[0]) if target_type == "x" else (x[0], target)
        x1, y1, x2, y2 = x[0], y[0], x[1], y[1]

    if target_type == "x":
        if x1 == x2:
            return target, y1
        return target, y1 + (y2 - y1) * (target - x1) / (x2 - x1)
    if y1 == y2:
        return x1, target
    return x1 + (x2 - x1) * (target - y1) / (y2 - y1), target

    def compute_cumulative_curve(
    displacement: np.ndarray,
    load: np.ndarray,
    markers: Optional[np.ndarray] = None,
    *,
    return_index: bool = False,
) -> Union[Tuple[np.ndarray, np.ndarray], Tuple[np.ndarray, np.ndarray, np.ndarray]]:
    """繰り返し載荷の荷重-変位曲線から累積曲線を計算します。

    各サイクルの正側（荷重 >= 0）の部分を、荷重 0 まで延長したうえで変位方向につなげます。

    Args:
        displacement: 変位の配列
        load: 荷重の配列
        markers: サイクル番号の配列（省略時は `compute_cycle_markers(load)`）
        return_index: 各点の元の行 index も返すか（荷重 0 まで延長した点は元の行がないため -1）

    Returns:
        累積曲線の (変位, 荷重)。return_index=True なら (変位, 荷重, 元の行 index)

    Examples:
        >>> from tascpy.numeric.hysteresis import compute_cumulative_curve
        >>> import numpy as np
        >>> disp = np.array([0.0, 1.0, 0.0, -1.0, 0.0, 2.0, 0.0])
        >>> load = np.array([0.0, 10.0, 0.0, -10.0, 0.0, 15.0, 0.0])
        >>> d_cum, p_cum = compute_cumulative_curve(disp, load)
    """
    _, groups = _cycle_groups(displacement, load, _markers_or_default(load, markers))

    p_cum: List[float] = []
    d_cum: List[float] = []
    rows: List[int] = []

    for cycle_load, cycle_disp, cycle_rows in groups:
        pos_idx = [
            j for j, (l, d) in enumerate(zip(cycle_load, cycle_disp))
            if not np.isnan(l) and not np.isnan(d) and l >= 0.0
        ]
        if not pos_idx:
            continue
        pos_loads = [cycle_load[j] for j in pos_idx]
        pos_displacements = [cycle_disp[j] for j in pos_idx]

        x_s, y_s = _extend_edge(pos_displacements, pos_loads, 0.0, "y", "start")
        x_e, y_e = _extend_edge(pos_displacements, pos_loads, 0.0, "y", "end")

        d_offset = d_cum[-1] - x_s if d_cum else 0.0

        p_cum.extend([y_s] + pos_loads + [y_e])
        d_cum.extend(x + d_offset for x in [x_s] + pos_displacements + [x_e])
        rows.extend([-1] + [cycle_rows[j] for j in pos_idx] + [-1])

    curve = (np.asarray(d_cum, dtype=float), np.asarray(p_cum, dtype=float))
    return (*curve, np.asarray(rows, dtype=np.int64)) if return_index else curve

    def compute_cycle_markers(values: np.ndarray, step: float = 0.5) -> np.ndarray:
    """値の符号反転からサイクル番号を数えます。

    0 と欠損（NaN）は符号を持たない点として飛ばし、直前の有効な値と符号が変わった位置で
    step だけ数を進めます（step=0.5 なら正負の往復で 1 サイクル）。サイクル番号は 1 から始まります。

    Args:
        values: 符号の反転を見る配列（通常は荷重。欠損は NaN または None）
        step: 符号反転1回あたりの増分

    Returns:
        np.ndarray: サイクル番号の配列（整数）

    Examples:
        >>> from tascpy.numeric.hysteresis import compute_cycle_markers
        >>> import numpy as np
        >>> compute_cycle_markers(np.array([10, 5, -5, -10, 5, 10]), step=0.5)
        array([1, 1, 1, 1, 2, 2])
        >>> compute_cycle_markers(np.array([10, 0, -10, np.nan, 10]), step=0.5)
        array([1, 1, 1, 1, 2])
    """
    sign = np.nan_to_num(np.sign(np.asarray(values, dtype=float)))
    nonzero = np.flatnonzero(sign)
    flips = np.zeros(len(sign))
    flips[nonzero[1:]] = sign[nonzero[1:]] != sign[nonzero[:-1]]
    return np.floor(1 + step * np.cumsum(flips)).astype(int)

    def compute_secant_stiffness(displacement: np.ndarray, load: np.ndarray) -> float:
    """1サイクルの割線剛性（最大荷重点と最小荷重点を結ぶ直線の傾き）を計算します。

    荷重・変位の両方が有効な点が2点未満、または2点の変位が同じなら NaN です。

    Examples:
        >>> from tascpy.numeric.hysteresis import compute_secant_stiffness
        >>> import numpy as np
        >>> compute_secant_stiffness(np.array([0, 1, 0, -1, 0]), np.array([0, 100, 0, -100, 0]))
        100.0
    """
    d, p = _valid_pairs(displacement, load)
    if len(p) < 2:
        return np.nan
    i_max, i_min = int(np.argmax(p)), int(np.argmin(p))
    delta = d[i_max] - d[i_min]
    if abs(delta) < 1e-9:
        return np.nan
    return float((p[i_max] - p[i_min]) / delta)

    def compute_skeleton_curve(
    displacement: np.ndarray,
    load: np.ndarray,
    markers: Optional[np.ndarray] = None,
    has_decrease: bool = False,
    decrease_type: str = "envelope",
    *,
    return_index: bool = False,
) -> Union[Tuple[np.ndarray, np.ndarray], Tuple[np.ndarray, np.ndarray, np.ndarray]]:
    """繰り返し載荷の荷重-変位曲線からスケルトン曲線を計算します。

    最大荷重のサイクルまでは、各サイクルで荷重がそれまでの最大を更新した点をつなぎます。
    2サイクル目以降は、そのサイクルで最初に最大を更新した点が、前のサイクルまでの包絡線の終端を
    延長した直線上に来るよう変位をずらします（ずらし量はサイクルごとに1回だけ決め、1サイクル目はずらさない）。

    曲線の各点は元データのいずれかの行で、荷重はその行の値、変位はその行の値にずらし量を足したものです。

    Args:
        displacement: 変位の配列
        load: 荷重の配列
        markers: サイクル番号の配列（省略時は `compute_cycle_markers(load)`）
        has_decrease: 最大荷重のサイクル以降の減少部分も含めるか
        decrease_type: 減少部分の取り方 ('envelope', 'continuous_only', 'both')
        return_index: 各点の元の行 index も返すか

    Returns:
        スケルトン曲線の (変位, 荷重)。return_index=True なら (変位, 荷重, 元の行 index)

    Examples:
        >>> from tascpy.numeric.hysteresis import compute_skeleton_curve
        >>> import numpy as np
        >>> disp = np.array([0.0, 1.0, 0.0, -1.0, 0.0, 2.0, 0.0])
        >>> load = np.array([0.0, 10.0, 0.0, -10.0, 0.0, 15.0, 0.0])
        >>> compute_skeleton_curve(disp, load)
        (array([1., 2.]), array([10., 15.]))
        >>> compute_skeleton_curve(disp, load, return_index=True)[2]
        array([1, 5])
    """
    if decrease_type not in ("envelope", "continuous_only", "both"):
        raise ValueError("decrease_typeは'envelope', 'continuous_only', 'both'のいずれかである必要があります")

    rows: List[int] = []
    p_ske: List[float] = []
    d_ske: List[float] = []

    def append(row: int, p: float, d: float, skip_duplicate: bool = False) -> None:
        if skip_duplicate and p_ske and p == p_ske[-1] and d == d_ske[-1]:
            return
        rows.append(row)
        p_ske.append(p)
        d_ske.append(d)

    markers = _markers_or_default(load, markers)
    keys, groups = _cycle_groups(displacement, load, markers)
    if len(keys) > 0:
        marker_arr = np.asarray(markers, dtype=float)
        has_marker = ~np.isnan(marker_arr)
        max_marker = marker_arr[has_marker][np.nanargmax(np.asarray(load, dtype=float)[has_marker])]
        p_max = 0.0
        d_offset = 0.0

        # 増加部分（最大荷重のサイクルまで）
        for key, (cycle_load, cycle_disp, cycle_rows) in zip(keys, groups):
            if key > max_marker:
                break
            d_offset = 0.0
            offset_fixed = False

            for i in range(len(cycle_load)):
                if not np.isnan(cycle_load[i]) and cycle_load[i] > p_max:
                    p_max = cycle_load[i]
                    # ずらし量はサイクルで最初に最大荷重を超えた点で、前のサイクルまでの包絡線から1回だけ決める
                    if not offset_fixed:
                        if len(p_ske) >= 2:
                            x, _ = _extend_edge(d_ske, p_ske, cycle_load[i], "y", "end")
                            d_offset = x - cycle_disp[i]
                        offset_fixed = True
                    append(cycle_rows[i], cycle_load[i], cycle_disp[i] + d_offset)

        # 減少部分（最大荷重のサイクル以降）。最大荷重のサイクルは増加部分と重なるため、直前と同じ点は追加しない
        if has_decrease:
            for key, (cycle_load, cycle_disp, cycle_rows) in zip(keys, groups):
                if key < max_marker:
                    continue

                valid_indices = [
                    i for i, (l, d) in enumerate(zip(cycle_load, cycle_disp))
                    if not np.isnan(l) and not np.isnan(d)
                ]
                if not valid_indices:
                    continue

                valid_load = [cycle_load[i] for i in valid_indices]
                valid_disp = [cycle_disp[i] for i in valid_indices]
                load_max_idx = valid_indices[valid_load.index(max(valid_load))]
                disp_max_idx = valid_indices[valid_disp.index(max(valid_disp))]

                if decrease_type == "envelope" or (decrease_type == "both" and load_max_idx == disp_max_idx):
                    append(cycle_rows[disp_max_idx], cycle_load[disp_max_idx], cycle_disp[disp_max_idx] + d_offset,
                           skip_duplicate=True)
                elif load_max_idx != disp_max_idx:
                    for i in range(min(load_max_idx, disp_max_idx), max(load_max_idx, disp_max_idx) + 1):
                        if np.isnan(cycle_load[i]) or np.isnan(cycle_disp[i]):
                            continue
                        append(cycle_rows[i], cycle_load[i], cycle_disp[i] + d_offset, skip_duplicate=True)

    curve = (np.asarray(d_ske, dtype=float), np.asarray(p_ske, dtype=float))
    return (*curve, np.asarray(rows, dtype=np.int64)) if return_index else curve

    def compute_rosette_strains(
    e1: np.ndarray,
    e2: np.ndarray,
    e3: np.ndarray,
    rosette_type: str = "rectangular",
    angle_offset: float = 0.0
) -> Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """ロゼットひずみゲージの値から主ひずみを計算します。

    Args:
        e1: 第1ゲージのひずみ配列
        e2: 第2ゲージのひずみ配列
        e3: 第3ゲージのひずみ配列
        rosette_type: ロゼットの種類（'rectangular': 0°/45°/90°、'delta': 0°/60°/120°）
        angle_offset: 第1ゲージの設置角度（度）

    Returns:
        Tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
            (最大主ひずみ, 最小主ひずみ, 最大せん断ひずみ, 主ひずみ方向角度)。
            角度は最大主ひずみの方向を、第1ゲージの方向から反時計回りに測った度数（angle_offset を加算）です。

    Examples:
        >>> from tascpy.numeric.rosette import compute_rosette_strains
        >>> import numpy as np
        >>> e1, e2, e3 = np.array([100]), np.array([200]), np.array([300])
        >>> compute_rosette_strains(e1, e2, e3, "rectangular")  # 最大主ひずみ 300 は第3ゲージ（90°）の方向
        (array([300.]), array([100.]), array([200.]), array([90.]))
    """
    if rosette_type.lower() == "rectangular":
        ex = e1
        ey = e3
        gamma_xy = 2 * e2 - (e1 + e3)
    elif rosette_type.lower() == "delta":
        ex = e1
        ey = (2 * e2 + 2 * e3 - e1) / 3.0
        gamma_xy = (2.0 / np.sqrt(3.0)) * (e2 - e3)
    else:
        raise ValueError(f"未対応のロゼットの種類です: {rosette_type}（'rectangular', 'delta' のいずれか）")

    center = (ex + ey) / 2.0
    radius = np.sqrt(((ex - ey) / 2.0) ** 2 + (gamma_xy / 2.0) ** 2)

    e_max = center + radius
    e_min = center - radius
    gamma_max = 2 * radius

    theta_rad = 0.5 * np.arctan2(gamma_xy, ex - ey)
    theta_deg = np.degrees(theta_rad)
    final_theta = theta_deg + angle_offset

    return e_max, e_min, gamma_max, final_theta

    def interpolate(points: Any, values: Any, targets: Any, method: str = "inverse_distance",
                power: float = 2.0) -> np.ndarray:
    """補間元の点の値から、補間先の座標での値を求めます

    - "inverse_distance": 逆距離加重 Σ(w·v)/Σw（w = 1/距離^power）。補間先が補間元の点と一致する場合はその点の値
    - "nearest": 最も近い補間元の点の値

    値が欠損（NaN）の補間元は、その列（時刻）の計算から除きます。有効な補間元がなければ NaN です。

    Args:
        points: 補間元の座標 (N, D)。D は 2 または 3
        values: 補間元の値 (N,)、または時刻ごとの値 (N, T)
        targets: 補間先の座標 (M, D)、または1点 (D,)
        method: 補間方法 ("inverse_distance", "nearest")
        power: 逆距離加重のべき指数

    Returns:
        np.ndarray: 補間値。形は (M,) / (M, T)、補間先が1点なら () / (T,)

    Raises:
        ValueError: 座標の次元が合わない場合、または未対応の method の場合

    Examples:
        >>> import numpy as np
        >>> from tascpy.numeric.spatial import interpolate
        >>> points = np.array([[0.0, 0.0], [10.0, 0.0]])
        >>> float(interpolate(points, [10.0, 20.0], [5.0, 0.0]))
        15.0
        >>> interpolate(points, [[10.0, 0.0], [20.0, 40.0]], [5.0, 0.0])
        array([15., 20.])
    """
    if method not in INTERPOLATION_METHODS:
        raise ValueError(f"サポートされていない補間方法: {method}（{', '.join(INTERPOLATION_METHODS)} のいずれか）")
    src = np.atleast_2d(np.asarray(points, dtype=float))
    raw_values = np.asarray(values, dtype=float)
    vals = raw_values.reshape(len(src), -1)
    raw_targets = np.asarray(targets, dtype=float)
    dst = np.atleast_2d(raw_targets)
    if src.shape[1] != dst.shape[1]:
        raise ValueError(f"補間元 ({src.shape[1]}次元) と補間先 ({dst.shape[1]}次元) の座標の次元が一致しません")

    distance = np.linalg.norm(dst[:, None, :] - src[None, :, :], axis=2)  # (M, N)
    valid = ~np.isnan(vals)  # (N, T)
    filled = np.where(valid, vals, 0.0)

    if method == "nearest":
        masked = np.where(valid.T[None, :, :], distance[:, None, :], np.inf)  # (M, T, N)
        nearest = np.argmin(masked, axis=2)
        result = np.take_along_axis(vals.T[None, :, :], nearest[..., None], axis=2)[..., 0]
        result = np.where(np.isinf(np.min(masked, axis=2)), np.nan, result)
    else:
        exact = distance < _EXACT
        with np.errstate(divide="ignore"):
            far_weight = np.where(exact, 0.0, 1.0 / distance**power)
        exact_weight = exact.astype(float)
        with np.errstate(invalid="ignore", divide="ignore"):
            far = (far_weight @ filled) / (far_weight @ valid)
            at_point = (exact_weight @ filled) / (exact_weight @ valid)
        result = np.where(np.isnan(at_point), far, at_point)

    result = result.reshape((len(dst),) + raw_values.shape[1:])
    return result[0] if raw_targets.ndim == 1 else result

    def kmeans(points: Any, n_clusters: int, max_iter: int = 100) -> np.ndarray:
    """点を k-means 法で n_clusters 個のクラスタに分けます（初期中心は乱数シード 42 で選ぶため結果は再現可能）

    Args:
        points: 座標の配列 (点の数, 次元)。整数やリストでもよい
        n_clusters: クラスタ数
        max_iter: 最大繰り返し回数

    Returns:
        np.ndarray: 各データ点のクラスタラベル

    Examples:
        >>> from tascpy.numeric.spatial import kmeans
        >>> import numpy as np
        >>> points = np.array([[1, 2], [1, 4], [1, 0], [10, 2], [10, 4], [10, 0]])
        >>> kmeans(points, 2)
        array([0, 0, 0, 1, 1, 1])
    """
    points = np.asarray(points, dtype=float)
    n_samples, _ = points.shape

    rng = np.random.default_rng(42)
    centers = points[rng.choice(n_samples, n_clusters, replace=False)]

    labels = np.zeros(n_samples, dtype=int)

    for _ in range(max_iter):
        old_labels = labels.copy()

        for i in range(n_samples):
            distances = np.sqrt(((points[i] - centers) ** 2).sum(axis=1))
            labels[i] = np.argmin(distances)

        if np.all(old_labels == labels):
            break

        for j in range(n_clusters):
            mask = labels == j
            if mask.sum() > 0:
                centers[j] = points[mask].mean(axis=0)

    return labels
