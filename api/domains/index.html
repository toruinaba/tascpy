    class LoadDisplacementCollection(ColumnCollection):
    """荷重と変位の列を指定したコレクション

    Examples:
        >>> ld_col = LoadDisplacementCollection(
        ...     {"P": [0.0, 10.0], "d": [0.0, 1.0]}, units={"P": "kN", "d": "mm"},
        ...     load_column="P", displacement_column="d",
        ... )
        >>> ld_col.load.unit
        'kN'
    """

    domain = "load_displacement"

    def __init__(self, columns: Optional[Mapping[str, Any]] = None, *, load_column: str,
                 displacement_column: str, **kwargs: Any) -> None:
        super().__init__(columns, **kwargs)
        self._load_column = load_column
        self._displacement_column = displacement_column
        self._validate()

    @classmethod
    def from_collection(cls, collection: ColumnCollection, load_column: str, displacement_column: str) -> Self:
        """既存のコレクションから、荷重列・変位列を指定して作成する

        Raises:
            ValueError: 指定した列が存在しない、または数値でない場合
        """
        new = collection._as_class(cls)
        new._load_column = load_column
        new._displacement_column = displacement_column
        new._validate()
        return new

    def _validate(self) -> None:
        for role, name in (("load_column", self._load_column), ("displacement_column", self._displacement_column)):
            if name not in self:
                raise ValueError(f"{role} に指定した列 '{name}' が存在しません")
            if not self[name].is_numeric:
                raise ValueError(f"{role} に指定した列 '{name}' は数値列ではありません")

    def _domain_config(self) -> Dict[str, Any]:
        return {"load_column": self._load_column, "displacement_column": self._displacement_column}

    @classmethod
    def _from_config(cls, collection: ColumnCollection, config: Mapping[str, Any]) -> Self:
        return cls.from_collection(collection, **config)

    def _apply_rename(self, mapping: Mapping[str, str]) -> None:
        self._load_column = mapping.get(self._load_column, self._load_column)
        self._displacement_column = mapping.get(self._displacement_column, self._displacement_column)

    @property
    def ops(self) -> "LoadDisplacementOps":
        """荷重-変位ドメインの操作オブジェクト"""
        from .ops import LoadDisplacementOps

        return LoadDisplacementOps(self)

    @property
    def plot(self) -> "LoadDisplacementPlotter":
        """荷重-変位データのプロッター"""
        from .plotter import LoadDisplacementPlotter

        return LoadDisplacementPlotter(self)

    @property
    def load_column(self) -> str:
        return self._load_column

    @property
    def displacement_column(self) -> str:
        return self._displacement_column

    @property
    def load(self) -> Column:
        return self[self._load_column]

    @property
    def displacement(self) -> Column:
        return self[self._displacement_column]

    @property
    def load_data(self) -> np.ndarray:
        """荷重の配列（欠損は NaN）"""
        return self.load.values

    @property
    def displacement_data(self) -> np.ndarray:
        """変位の配列（欠損は NaN）"""
        return self.displacement.values

    @property
    def valid_data_mask(self) -> np.ndarray:
        """荷重・変位がともに欠損でない行のマスク"""
        return ~(np.isnan(self.load_data) | np.isnan(self.displacement_data))

    @property
    def valid_data(self) -> Tuple[np.ndarray, np.ndarray]:
        """有効な行の (変位, 荷重)"""
        mask = self.valid_data_mask
        return self.displacement_data[mask], self.load_data[mask]

    def __repr__(self) -> str:
        return (super().__repr__()[:-1]
                + f" load={self._load_column!r} displacement={self._displacement_column!r}>")

    class LoadDisplacementOps(CoreOps[LoadDisplacementCollection]):
    """荷重-変位コレクションの操作（汎用操作も全て利用可能）"""

    def _cycle_markers(self, cycle_column: Optional[str]) -> np.ndarray:
        """サイクル番号の配列（cycle_column の省略時は荷重の符号反転から数える。mark_cycles の既定と同じ）"""
        if cycle_column is None:
            return compute_cycle_markers(self._c.load_data)
        return to_float_array(self._column_values(cycle_column))

    def _ld_arrays(self) -> Tuple[np.ndarray, np.ndarray]:
        return self._c.displacement_data, self._c.load_data

    def _ld_unit(self, sep: str) -> Optional[str]:
        return self._compound_unit(self._c.load_column, self._c.displacement_column, sep)

    def stiffness(self, *, range_start: float = 0.2, range_end: float = 0.8,
                  method: str = "linear_regression") -> ScalarResult:
        """最大荷重に対する比率で指定した範囲の剛性を求めます（単位は "荷重/変位"）

        変位が変化しない区間があると計算に失敗するため、事前に
        `remove_consecutive_duplicates_across()` で重複を除くことを推奨します。

        単調載荷の曲線を前提にしています。繰り返し載荷のデータにそのまま使うと、範囲に入る除荷途中の点も
        計算に含まれ、エラーにならずに誤った値になります。`to_skeleton_curve()` でスケルトン曲線にするか、
        最初の載荷区間を抜き出してから求めてください。

        Args:
            range_start: 範囲の開始比率
            range_end: 範囲の終了比率
            method: "linear_regression" または "secant"

        Examples:
            >>> k = ld.ops.stiffness(range_start=0.1, range_end=0.8)
            >>> round(k.value, 1), k.unit
            (40.0, 'kN/mm')

            繰り返し載荷のデータ（`cyclic`）では、スケルトン曲線から求めます。

            >>> round(cyclic.ops.stiffness(range_start=0.1, range_end=0.5).value, 1)  # 生データのまま（誤り）
            8.6
            >>> round(cyclic.ops.to_skeleton_curve().stiffness(range_start=0.1, range_end=0.5).value, 1)
            40.0
        """
        value = compute_stiffness(*self._ld_arrays(), range_start=range_start, range_end=range_end, method=method)
        return ScalarResult(name="stiffness", value=value, unit=self._ld_unit("/"))

    def yield_point(self, *, method: str = "offset", offset_value: float = 0.002, range_start: float = 0.1,
                    range_end: float = 0.3, factor: float = 0.33, fail_silently: bool = False) -> PointResult:
        """降伏点を求めます（x: 変位, y: 荷重。metadata に is_valid と計算条件）

        初期剛性を最大荷重の range_start〜range_end の範囲から求め、
        - "offset": 初期剛性線を offset_value だけ平行移動した直線との交点
        - "general": 接線剛性が初期剛性 × factor を下回る点
        を降伏点とします。

        単調載荷の曲線を前提にしています（`stiffness` と同じ）。繰り返し載荷のデータでは、
        `to_skeleton_curve()` でスケルトン曲線にしてから求めてください。

        Examples:
            >>> pt = ld.ops.yield_point(offset_value=0.5, range_start=0.1, range_end=0.8)
            >>> round(pt.x, 2), round(pt.y, 1), pt.x_unit, pt.y_unit
            (3.0, 100.0, 'mm', 'kN')

            繰り返し載荷のデータ（`cyclic`）では、スケルトン曲線から求めます。

            >>> pt = cyclic.ops.to_skeleton_curve().yield_point(offset_value=0.5, range_start=0.1, range_end=0.5)
            >>> round(pt.x, 2), round(pt.y, 1)
            (3.06, 102.2)
        """
        outcome = compute_yield_point(
            *self._ld_arrays(), method=method, offset_value=offset_value, range_start=range_start,
            range_end=range_end, factor=factor, fail_silently=fail_silently,
        )
        return self._point("yield_point", outcome, x_unit=self._c.displacement.unit, y_unit=self._c.load.unit)

    def mark_cycles(self, *, column: Optional[str] = None, step: float = 0.5, result_column: str,
                    unit: Optional[str] = None, ch: Optional[str] = None) -> Self:
        """荷重の符号反転に基づくサイクル番号の列を追加します

        サイクルを使う操作（`to_skeleton_curve` など）は cycle_column の省略時に同じ規則でサイクルを数えるため、
        この列は既定と異なる数え方をしたいときや、サイクル番号を列として使いたいときに作ります。

        Args:
            column: 判定に使う列（省略時は荷重列）
            step: 符号反転1回あたりのサイクル番号の増分（0.5 なら正負の往復で 1 サイクル）
        """
        source = column or self._c.load_column
        markers = compute_cycle_markers(to_float_array(self._column_values(source)), step=step)
        return self._with_column(result_column, markers.tolist(), unit, ch, default_unit="")

    def split_by_cycles(self: LS, *, cycle_column: Optional[str] = None) -> OpsList[LS]:
        """サイクル番号ごとに分割します（cycle_column の省略時は荷重の符号反転から数える）"""
        return self.split_by_integers(self._cycle_markers(cycle_column))

    def split_by_reversals(self: LS, *, column: Optional[str] = None, ratio: float = 0.05,
                           min_amplitude: float = 0.0) -> OpsList[LS]:
        """反転点（正側・負側の極値）から次の反転点までの半サイクルごとに分割します

        `split_by_cycles` が荷重の符号反転（0 をまたぐ位置）で切るのに対し、こちらはピークで切ります。
        除荷から逆方向の再載荷までが1つの区間に入るので、`shift_origin()` で区間の始点を原点にすれば、
        `yield_point` / `stiffness` で再降伏点や除荷剛性を半サイクルごとに求められます。

        隣り合う区間は境界の反転点を共有します（その点は両方の区間に入ります）。
        各要素のキー（`OpsList.keys`）は半サイクルの番号 1, 2, ... です。

        Args:
            column: 反転を見る列（省略時は変位列）
            ratio: 反転を確定する戻り量の、直前の反転点からの片振幅に対する比率
            min_amplitude: 反転とみなす最小の片振幅（column の単位。立ち上がりのノイズ除け）

        Examples:
            `cyclic` は 0 → +2 → -2 → +4 → 0 mm の載荷です。

            >>> parts = cyclic.ops.split_by_reversals()
            >>> parts.keys, [c["変位"].values[[0, -1]].tolist() for c in parts.end()]
            ([1, 2, 3, 4], [[0.0, 2.0], [2.0, -2.0], [-2.0, 4.0], [4.0, 0.0]])

            区間ごとに始点を原点へ移して降伏点を求めます。降伏点は移動後の座標なので、
            移動量（metadata の "origin_load"）を足すと元の荷重になります。
            弾性範囲で折り返した区間は降伏点が求まらず、欠損になります。

            >>> shifted = parts.map(lambda o: o.shift_origin())
            >>> points = shifted.collect(lambda o: o.yield_point(offset_value=0.5, range_start=0.1, range_end=0.5,
            ...                                                  fail_silently=True))
            >>> origin = shifted.collect(lambda o: o.end().metadata["origin_load"])
            >>> table = points.to_table(name="yield").join(origin.to_table(name="origin_load")).end()
            >>> (table["yield_y"].values + table["origin_load"].values).round(1)
            array([  nan,   nan, 102.2,   nan])
        """
        source = self._c.displacement_column if column is None else column
        values = to_float_array(self._column_values(source))
        reversals = compute_reversal_indices(values, ratio=ratio, min_amplitude=min_amplitude).tolist()
        bounds = [0, *reversals, len(self._c) - 1] if len(self._c) else []
        groups = [np.arange(a, b + 1) for a, b in zip(bounds[:-1], bounds[1:])]
        return self._split(groups, list(range(1, len(groups) + 1)))

    def split_by_excursions(self: LS, *, side: Side = "positive",
                            cycle_column: Optional[str] = None) -> OpsList[LS]:
        """荷重の片側の区間（荷重 0 の点から荷重 0 の点まで）ごとに分割します（荷重 0 基準）

        各区間は元データの行（全列・元のステップ）に、補間で求めた荷重 0 の点の行を前後に加えたものです。
        加えた行は、荷重が 0・変位が補間値（データの先頭では最初の2行の延長）で、ステップは前後の行のステップの
        直線補間（小数。延長した点は行の間にないので欠損）、他の列は欠損です
        （端の行の荷重がちょうど 0 の場合と、データの末尾で荷重 0 に届いていない端には加えません）。
        区間の取り方は `to_cumulative_curve` と同じです（`tascpy.numeric.hysteresis.compute_excursions`）。
        各要素のキーは区間の番号 1, 2, ... です。

        `OpsList.chain` で変位列をつなぐと累積曲線になり、`concat()` は `to_cumulative_curve` と同じ曲線です。

        Args:
            side: "positive"（荷重 >= 0 の区間）または "negative"（荷重 <= 0 の区間）
            cycle_column: サイクル番号の列（省略時は荷重の符号反転から数える。番号が変わる位置でも区切る）

        Examples:
            `cyclic` の正側の区間は、0 → +2 → 0 mm と、0 → +4 → 1.35 mm（荷重 0 の点）の2つです。

            >>> parts = cyclic.ops.split_by_excursions()
            >>> parts.keys, [c["変位"].values[[0, -1]].round(2).tolist() for c in parts.end()]
            ([1, 2], [[0.0, 0.0], [0.0, 1.35]])
            >>> float(parts[1].end().step[-1])  # 補った荷重 0 の点のステップは、前後の行の間の小数
            29.3

            区間をつなぐと累積曲線になります。

            >>> curve = parts.chain("変位").concat().end()
            >>> np.allclose(curve.displacement_data, cyclic.ops.to_cumulative_curve().end().displacement_data)
            True
        """
        excursions = compute_excursions(*self._all_cycle_arrays(cycle_column), side=side)
        parts = [self._next(self._assemble(
            ([] if e.start is None else [(-1, e.start, 0.0, e.start_position)])
            + [(int(r), np.nan, np.nan, float(r)) for r in e.rows]
            + ([] if e.end is None else [(-1, e.end, 0.0, e.end_position)])
        )) for e in excursions]
        return OpsList(parts, list(range(1, len(parts) + 1)))

    def _assemble(self, entries: list) -> LoadDisplacementCollection:
        """(行 index, 変位, 荷重, 行の位置) の並びのコレクション

        index が -1 の点は、変位・荷重と、行の位置で補間したステップ（小数。位置が NaN なら欠損）だけを持つ行
        （他の列は欠損）にする。点の行を加えるとステップの列は Float64 になる（ステップが数値でなければ欠損のまま）。
        元の行の点は、変位が NaN でなければその値に置き換える（`clip_load(close_gaps=True)` でずらした変位など）。
        """
        c = self._c
        added = [(x, y, pos) for row, x, y, pos in entries if row < 0]
        if not added:
            return self._with_displacement(c.take([row for row, *_ in entries]), entries)
        numeric_step = c.df["step"].dtype.is_numeric()
        dtype = pl.Float64 if numeric_step else c.df["step"].dtype
        steps = (interpolate_positions(c.step.astype(float), [pos for _, _, pos in added]) if numeric_step
                 else [None] * len(added))
        frames, pending = [], []

        def flush() -> None:
            if pending:
                frames.append(c.take(list(pending)).df.with_columns(pl.col("step").cast(dtype)))
                pending.clear()

        k = 0
        for row, x, y, _ in entries:
            if row >= 0:
                pending.append(row)
                continue
            flush()
            step = steps[k]
            k += 1
            frames.append(pl.DataFrame({
                "step": pl.Series([None if step is None or np.isnan(step) else float(step)], dtype=dtype),
                c.load_column: [float(y)], c.displacement_column: [float(x)],
            }))
        flush()
        return self._with_displacement(
            c._replace(df=pl.concat(frames, how="diagonal_relaxed").select(c.df.columns)), entries)

    def _with_displacement(self, collection: LoadDisplacementCollection, entries: list) -> LoadDisplacementCollection:
        """点の並びの変位（NaN でないもの）で変位列を置き換える（NaN の点は元の値のまま）"""
        x = np.array([e[1] for e in entries], dtype=float)
        if np.isnan(x).all():
            return collection
        name = collection.displacement_column
        values = np.where(np.isnan(x), collection.displacement_data, x)
        return collection._replace(df=collection.df.with_columns(pl.Series(name, values, dtype=pl.Float64)))

    def _entries(self, d: np.ndarray, p: np.ndarray, positions: np.ndarray) -> list:
        """numeric の (変位, 荷重, 行の位置) を `_assemble` の点の並びにする

        整数の位置で荷重・変位が欠損でない行は元の行の点、それ以外（小数・NaN の位置、欠損の行に重なる位置）は補った点。
        """
        valid = self._c.valid_data_mask
        entries = []
        for x, y, pos in zip(d.tolist(), p.tolist(), positions.tolist()):
            row = int(pos) if not np.isnan(pos) and pos == int(pos) and valid[int(pos)] else -1
            entries.append((row, x, y, pos))
        return entries

    def clip_load(self, *, lower: Optional[float] = None, upper: Optional[float] = None,
                  close_gaps: bool = False) -> Self:
        """荷重が lower 以上・upper 以下の行を残し、範囲の境界をまたぐ位置に境界の点の行を補間で加えます

        加えた行は、荷重が境界の値・変位が補間値で、ステップは前後の行のステップの直線補間（小数）、他の列は欠損です
        （`split_by_excursions` の荷重 0 の点と同じ扱い）。荷重・変位が欠損の行は除きます。`scan` と組み合わせると、区間ごとに「それまでの最大荷重を超える部分」
        （スケルトン部分）や「それ以下の部分」（バウシンガー部分）を取り出せます。

        範囲の外に出て戻る曲線では、既定では範囲内の部分を元の変位のまま並べます（外に出ていた間の変位が残ります）。
        close_gaps=True なら、戻った部分の変位を、出た点に続くようにずらして詰めます（バウシンガー部分の取り出しなど）。

        Args:
            lower: 荷重の下限（省略時は下限なし）
            upper: 荷重の上限（省略時は上限なし）
            close_gaps: 範囲の外に出ていた間の変位を詰めるか

        Examples:
            >>> part = ld.ops.clip_load(lower=60.0).end()
            >>> part["荷重"].values, part["変位"].values
            (array([ 60.,  80., 100., 110., 115.]), array([1.5, 2. , 3. , 5. , 8. ]))
            >>> float(part.step[0])  # 補った境界の点のステップは、前後の行の間の小数
            2.5

            区間ごとに、それまでの最大荷重以上の部分を取り出してつなぐとスケルトン曲線になります
            （`to_skeleton_curve` は、これからつなぎ目で重なる点を除いたものです）。耐力を更新しない区間は空になります。

            >>> def above(p_max, o):
            ...     return o.clip_load(lower=p_max), max(p_max, float(np.nanmax(o.end().load_data)))
            >>> loading = cyclic.ops.split_by_excursions().map(lambda o: o.until_peak(column="荷重"))
            >>> curve = loading.scan(above, 0.0).chain("変位").concat().end()
            >>> curve["変位"].values, curve["荷重"].values
            (array([0. , 0.5, 1. , 1.5, 2. , 2. , 2.5, 3. , 3.5, 4. ]), array([  0.,  20.,  40.,  60.,  80.,  80., 100., 102., 104., 106.]))
        """
        d, p, positions = compute_load_clip(self._c.displacement_data, self._c.load_data, lower=lower, upper=upper,
                                            close_gaps=close_gaps)
        return self._next(self._assemble(self._entries(d, p, positions)))

    def unloading_stiffness(self) -> ScalarResult:
        """区間（`split_by_excursions` の要素など）の除荷剛性を求めます（単位は "荷重/変位"）

        変位のピーク（始点から最も離れる点。除荷の始まり）から、最後の点までの割線剛性です。最後の点の荷重が 0 でない
        （除荷して荷重 0 に戻っていない）場合は NaN です。計算の詳細は
        `tascpy.numeric.hysteresis.compute_unloading_stiffness` を参照してください。

        Examples:
            >>> k = cyclic.ops.split_by_excursions()[1].unloading_stiffness()
            >>> round(k.value, 1), k.unit
            (40.0, 'kN/mm')
        """
        value = compute_unloading_stiffness(self._c.displacement_data, self._c.load_data)
        return ScalarResult("unloading_stiffness", value, unit=self._ld_unit("/"))

    def append_unloading(self, stiffness: Union[float, ScalarResult]) -> Self:
        """最後の有効な点から、剛性 stiffness で荷重 0 まで戻した点の行を末尾に加えます

        加えた行は、荷重が 0・変位が「最後の点の変位 − 荷重 / 剛性」で、ステップと他の列は欠損です
        （データの行の間にある点ではないため）。最後の点の荷重が
        0 の場合と、剛性が NaN・0 の場合は何もしません。区間から一部を切り出した部分（`clip_load` の結果など）を、
        除荷剛性で戻してから `chain` でつなぐときに使います（`to_bauschinger_curve` はこの形です）。

        Args:
            stiffness: 剛性（荷重/変位）。`unloading_stiffness()` の結果も渡せます

        Examples:
            >>> part = ld.ops.until_peak().clip_load(upper=100.0).append_unloading(40.0).end()
            >>> part["変位"].values[-2:], part["荷重"].values[-2:]
            (array([3. , 0.5]), array([100.,   0.]))
        """
        value = float(stiffness.value if isinstance(stiffness, ScalarResult) else stiffness)
        c = self._c
        d, p = c.displacement_data, c.load_data
        valid = np.flatnonzero(~(np.isnan(d) | np.isnan(p)))
        if len(valid) == 0 or p[valid[-1]] == 0 or not np.isfinite(value) or value == 0:
            return self
        x, y = float(d[valid[-1]]), float(p[valid[-1]])
        entries = [(i, np.nan, np.nan, float(i)) for i in range(len(c))] + [(-1, x - y / value, 0.0, np.nan)]
        return self._next(self._assemble(entries))

    def mark_plastic(self, *, range_start: float = 0.1, range_end: float = 0.3, direction: Direction = "both",
                     result_column: str, unit: Optional[str] = None, ch: Optional[str] = None) -> Self:
        """載荷区間の始点からの塑性変形を列として追加します（`split_by_reversals` の区間に使います）

        区間の最初の有効な点を原点とし、その曲線の剛性 K を range_start〜range_end の範囲から求めて
        （`shift_origin().yield_point()` の初期剛性と同じ求め方。範囲は `yield_point` とそろえます）、
        載荷の向きへの塑性変形 Δd − ΔP/K のそれまでの最大値（負なら 0）を求めます。値は始点で 0 で、単調に増えます。
        単位は省略時に変位列の単位です。剛性を求められない場合は、警告を出して 0 にします。

        区間ごとに求めた列を `OpsList.chain` でつなぐと、累積塑性変形になります。
        計算の詳細は `tascpy.numeric.hysteresis.compute_plastic_deformation` を参照してください。

        Args:
            range_start: 剛性を求める範囲の開始（ΔP の最大値に対する割合）
            range_end: 剛性を求める範囲の終了（ΔP の最大値に対する割合）
            direction: "both"、または "positive" / "negative"（載荷の向きが異なる区間は 0）

        Examples:
            `cyclic` は -2 → +4 mm の半サイクルで 1.35 mm 塑性化し、他の半サイクルは弾性です。

            >>> halves = cyclic.ops.split_by_reversals().map(
            ...     lambda o: o.mark_plastic(range_start=0.1, range_end=0.5, result_column="塑性"))
            >>> [round(float(c["塑性"].values[-1]), 2) for c in halves.end()]
            [0.0, 0.0, 1.35, 0.0]

            つなぐと累積塑性変形です。半サイクルの先頭の値が、その反転に入るまでの累積です。

            >>> cum = halves.chain("塑性")
            >>> [round(float(c["塑性"].values[0]), 2) for c in cum.end()]
            [0.0, 0.0, 0.0, 1.35]
        """
        c = self._c
        try:
            values = compute_plastic_deformation(c.displacement_data, c.load_data, range_start=range_start,
                                                 range_end=range_end, direction=direction)
        except ValueError as e:
            if "direction" in str(e):
                raise
            warnings.warn(f"塑性変形を 0 にしました（{e}）", UserWarning, stacklevel=2)
            values = np.where(c.valid_data_mask, 0.0, np.nan)
        return self._with_column(result_column, values, unit, ch, default_unit=c.displacement.unit)

    def until_peak(self, *, column: Optional[str] = None) -> Self:
        """先頭から、列の値が最初の有効な値から最も離れる行（ピーク）までの行を残します

        除荷を含む区間（`split_by_excursions` の区間など）から載荷部分を取り出し、`yield_point` / `stiffness` などの
        単調な曲線を前提にした計算に使うためのものです。ピークが複数の行にある場合は最初の行までです。

        Args:
            column: ピークを見る列（省略時は変位列）

        Raises:
            ValueError: 列に有効な値がない場合

        Examples:
            `cyclic` は 0 → +2 → -2 → ... mm の載荷なので、最初の正のピーク（+2 mm）ではなく、
            最も離れる +4 mm までを残します。

            >>> cyclic.ops.until_peak().end()["変位"].values[[0, -1]]
            array([0., 4.])

            区間ごとに使うと、各区間の載荷部分になります。

            >>> parts = cyclic.ops.split_by_excursions()
            >>> [c["変位"].values[[0, -1]].tolist() for c in parts.map(lambda o: o.until_peak()).end()]
            [[0.0, 2.0], [0.0, 4.0]]
        """
        name = self._c.displacement_column if column is None else column
        values = np.asarray(self._numeric(name), dtype=float)
        valid = np.flatnonzero(~np.isnan(values))
        if len(valid) == 0:
            raise ValueError(f"列 '{name}' に有効な値がありません")
        peak = int(np.nanargmax(np.abs(values - values[valid[0]])))
        return self._with_rows(np.arange(peak + 1))

    def shift_origin(self) -> Self:
        """最初の有効な点（荷重・変位とも欠損でない点）が原点になるよう、荷重と変位を平行移動します

        移動量は metadata の "origin_displacement" / "origin_load" に記録します（元の座標の値は
        移動後の値 + 移動量）。繰り返し呼ぶと移動量は累積します。
        `split_by_reversals` の区間に使うと、反転点を原点とした除荷・再載荷の曲線になります。

        Raises:
            ValueError: 荷重・変位とも欠損でない点がない場合

        Examples:
            >>> part = cyclic.ops.split_by_reversals()[2].shift_origin().end()
            >>> part["変位"].values[[0, -1]], part["荷重"].values[[0, -1]]
            (array([0., 6.]), array([  0., 186.]))
            >>> part["荷重"].unit
            'kN'
            >>> part.metadata["origin_displacement"], part.metadata["origin_load"]
            (-2.0, -80.0)
        """
        c = self._c
        d, p = c.displacement_data, c.load_data
        valid = np.flatnonzero(~(np.isnan(d) | np.isnan(p)))
        if len(valid) == 0:
            raise ValueError("荷重・変位とも欠損でない点がないため、原点を決められません")
        d0, p0 = float(d[valid[0]]), float(p[valid[0]])
        shifted = (c.with_column(c.displacement_column, d - d0).with_column(c.load_column, p - p0)
                   .with_metadata(origin_displacement=c.metadata.get("origin_displacement", 0.0) + d0,
                                  origin_load=c.metadata.get("origin_load", 0.0) + p0))
        return self._next(shifted)

    def mark_peaks_and_valleys(self, column: Optional[str] = None, distance: int = 1,
                               threshold: Optional[float] = None, *, result_column: str,
                               unit: Optional[str] = None, ch: Optional[str] = None) -> Self:
        """ピーク (1) / バレー (-1) / その他 (0) のマーカー列を追加します

        Args:
            column: 対象列（省略時は荷重列）
            distance: 隣接ピーク間の最小距離
            threshold: ピークとみなす閾値
        """
        values = self._c.load_data if column is None else to_float_array(self._column_values(column))
        flags = compute_peaks_and_valleys(values, distance=distance, threshold=threshold)
        return self._with_column(result_column, flags, unit, ch, default_unit="")

    def to_hysteresis_table(self, *, cycle_column: Optional[str] = None) -> CoreOps[ColumnCollection]:
        """サイクルごとの履歴エネルギーと最大・最小荷重/変位を集計した表を返します

        戻り値のステップはサイクル番号、列は energy（単位は "荷重*変位"）/ max_load / min_load / max_disp / min_disp です。
        """
        load_unit, disp_unit = self._c.load.unit, self._c.displacement.unit
        rows = self.split_by_cycles(cycle_column=cycle_column).collect(
            lambda o: compute_hysteresis_energy(*o._ld_arrays()))
        units = {"energy": self._ld_unit("*"), "max_load": load_unit, "min_load": load_unit,
                 "max_disp": disp_unit, "min_disp": disp_unit}
        columns = {name: [row[i] for row in rows] for i, name in enumerate(units)}
        return CoreOps(ColumnCollection(columns, step=rows.keys, units=units, metadata=self._c.metadata))

    def to_stiffness_table(self, *, cycle_column: Optional[str] = None) -> CoreOps[ColumnCollection]:
        """サイクルごとの割線剛性を集計した表を返します（列: stiffness）"""
        unit = self._ld_unit("/")

    def _assemble(self, entries: list) -> LoadDisplacementCollection:
    """(行 index, 変位, 荷重, 行の位置) の並びのコレクション

    index が -1 の点は、変位・荷重と、行の位置で補間したステップ（小数。位置が NaN なら欠損）だけを持つ行
    （他の列は欠損）にする。点の行を加えるとステップの列は Float64 になる（ステップが数値でなければ欠損のまま）。
    元の行の点は、変位が NaN でなければその値に置き換える（`clip_load(close_gaps=True)` でずらした変位など）。
    """
    c = self._c
    added = [(x, y, pos) for row, x, y, pos in entries if row < 0]
    if not added:
        return self._with_displacement(c.take([row for row, *_ in entries]), entries)
    numeric_step = c.df["step"].dtype.is_numeric()
    dtype = pl.Float64 if numeric_step else c.df["step"].dtype
    steps = (interpolate_positions(c.step.astype(float), [pos for _, _, pos in added]) if numeric_step
             else [None] * len(added))
    frames, pending = [], []

    def flush() -> None:
        if pending:
            frames.append(c.take(list(pending)).df.with_columns(pl.col("step").cast(dtype)))
            pending.clear()

    k = 0
    for row, x, y, _ in entries:
        if row >= 0:
            pending.append(row)
            continue
        flush()
        step = steps[k]
        k += 1
        frames.append(pl.DataFrame({
            "step": pl.Series([None if step is None or np.isnan(step) else float(step)], dtype=dtype),
            c.load_column: [float(y)], c.displacement_column: [float(x)],
        }))
    flush()
    return self._with_displacement(
        c._replace(df=pl.concat(frames, how="diagonal_relaxed").select(c.df.columns)), entries)

        >>> round(pt.x, 2), round(pt.y, 1), pt.x_unit, pt.y_unit
        (3.0, 100.0, 'mm', 'kN')

        繰り返し載荷のデータ（`cyclic`）では、スケルトン曲線から求めます。

        >>> pt = cyclic.ops.to_skeleton_curve().yield_point(offset_value=0.5, range_start=0.1, range_end=0.5)
        >>> round(pt.x, 2), round(pt.y, 1)
        (3.06, 102.2)
    """
    outcome = compute_yield_point(
        *self._ld_arrays(), method=method, offset_value=offset_value, range_start=range_start,
        range_end=range_end, factor=factor, fail_silently=fail_silently,
    )
    return self._point("yield_point", outcome, x_unit=self._c.displacement.unit, y_unit=self._c.load.unit)

    def _peak_load(x_values: Optional[np.ndarray], y_values: Optional[np.ndarray]) -> Optional[float]:
    """荷重・変位がともに有効な行のうち、絶対値が最大の荷重（なければ None）"""

    class CoordinateCollection(ColumnCollection):
    """列ごとに座標 (x, y, z) を持つコレクション

    座標は各列の `ColumnMeta.attrs["coordinates"]` に格納されるため、列の改名・選択に追従します。

    Examples:
        >>> coord_col = CoordinateCollection({"CH1": [100.0]}, coordinates={"CH1": {"x": 10.0, "y": 20.0}})
        >>> coord_col.get_column_coordinates("CH1")
        (10.0, 20.0, None)
    """

    domain = "coordinate"

    def __init__(self, columns: Optional[Mapping[str, Any]] = None, *,
                 coordinates: Optional[Coordinates] = None, **kwargs: Any) -> None:
        super().__init__(columns, **kwargs)
        configured = self._with_coordinate_map(coordinates)
        self._set(configured._df, configured._meta, configured._metadata, configured._results)

    @classmethod
    def from_collection(cls, collection: ColumnCollection, *, coordinates: Optional[Coordinates] = None) -> Self:
        return collection._as_class(cls)._with_coordinate_map(coordinates)

    def _with_coordinate_map(self, coordinates: Optional[Coordinates]) -> Self:
        if coordinates is None:
            return self
        if isinstance(coordinates, (str, Path)):
            coordinates = load_coordinates(coordinates)
        new = self
        for name, xyz in coordinates.items():
            if name in new:
                new = new.with_coordinates(name, x=xyz.get("x"), y=xyz.get("y"), z=xyz.get("z"))
        return new

    @property
    def ops(self) -> "CoordinateOps[Self]":
        """座標ドメインの操作オブジェクト"""
        from .ops import CoordinateOps

        return CoordinateOps(self)

    # ── 座標 ──────────────────────────────────

    def with_coordinates(self, column_name: str, x: Optional[float] = None, y: Optional[float] = None,
                         z: Optional[float] = None) -> Self:
        """列の座標を設定する（None の軸は既存の値を残す）"""
        if column_name not in self:
            raise ValueError(f"カラム '{column_name}' が見つかりません")
        coords = dict(self.meta[column_name].attrs.get(COORDINATES_KEY, {"x": None, "y": None, "z": None}))
        coords.update({k: v for k, v in {"x": x, "y": y, "z": z}.items() if v is not None})
        return self.with_attrs(column_name, **{COORDINATES_KEY: coords})

    def get_column_coordinates(self, column_name: str) -> Tuple[Optional[float], Optional[float], Optional[float]]:
        """列の座標 (x, y, z)（未設定の軸は None）"""
        if column_name not in self:
            raise ValueError(f"カラム '{column_name}' が見つかりません")
        coords = self.meta[column_name].attrs.get(COORDINATES_KEY, {})
        return coords.get("x"), coords.get("y"), coords.get("z")

    def get_columns_with_coordinates(self) -> List[str]:
        """いずれかの軸の座標が設定されている列名"""
        return [name for name in self.columns if any(v is not None for v in self.get_column_coordinates(name))]

    def load_coordinates(path: Union[str, Path]) -> Dict[str, Dict[str, Optional[float]]]:
    """座標ファイルを読み込む

    JSON は {列名: {"x": .., "y": .., "z": ..}}、CSV は column, x, y, z 列（大文字小文字は区別しない）の形式です。
    """
    path = Path(path)
    if not path.exists():
        raise FileNotFoundError(f"座標ファイルが見つかりません: {path}")
    suffix = path.suffix.lower()
    if suffix == ".json":
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    if suffix != ".csv":
        raise ValueError(f"サポートされていないファイル形式です: {suffix}")

    result: Dict[str, Dict[str, Optional[float]]] = {}
    with open(path, encoding="utf-8") as f:
        reader = csv.DictReader(f)
        header = {h.lower(): h for h in reader.fieldnames or []}
        if "column" not in header:
            return result
        for row in reader:
            coords: Dict[str, Optional[float]] = {}
            for axis in ("x", "y", "z"):
                raw = row.get(header[axis]) if axis in header else None
                if raw:
                    try:
                        coords[axis] = float(raw)
                    except ValueError:
                        coords[axis] = None
            result[row[header["column"]]] = coords
    return result

    def _sources(self, names: Optional[List[str]], use_z: bool) -> Tuple[List[str], np.ndarray, np.ndarray]:
        """補間元の (列名, 座標 (N, D), 値 (N, 行数))。座標が揃わない列・数値でない列は除く"""
        used, points = [], []
        for name in self._require_coordinates() if names is None else self._check_columns(names):
            x, y, z = self._c.get_column_coordinates(name)
            if x is None or y is None or (use_z and z is None) or not self._c[name].is_numeric:
                continue

            points.append([x, y, z] if use_z else [x, y])
        if not used:
            raise ValueError("補間に使用できる座標付きの数値列がありません")
        values = np.vstack([self._c[name].values for name in used])
        return used, np.array(points, dtype=float), values

        units = {self._unit_of(name) for name in names}
        return units.pop() if len(units) == 1 else None

    def interpolate_at_point(self, x: float, y: float, z: Optional[float] = None,
                             source_columns: Optional[List[str]] = None, method: str = "inverse_distance",
                             power: float = 2.0, *, result_column: str,
                             unit: Optional[str] = None) -> Self:
        """座標 (x, y[, z]) での値を行ごとに補間した列を追加します

        追加した列には補間点の座標が設定されます。単位は、補間元の単位がすべて同じならその単位、
        異なれば None です。z を指定した場合は z 座標を持つ列だけを補間元にします。

        Args:
            source_columns: 補間元の列（省略時は座標を持つ全列）
            method: "inverse_distance" または "nearest"
            power: 逆距離加重のべき指数
            result_column: 結果列名（既存列を指定すると上書き）

        Examples:
            >>> center = coord.ops.interpolate_at_point(5.0, 0.0, source_columns=["CH1", "CH2"], result_column="center").end()
            >>> center["center"].values, center.get_column_coordinates("center")
            (array([  0.,  75., 150., 225., 300., 375.]), (5.0, 0.0, None))
        """
        use_z = z is not None
        names, points, values = self._sources(source_columns, use_z)
        target = [x, y, z] if use_z else [x, y]
        result = interpolate(points, values, target, method=method, power=power)

        collection = self._c.with_column(
            result_column, result, unit=unit if unit is not None else self._common_unit(names),
            attrs={"interpolation": self._info(method, power, names)},
        ).with_coordinates(result_column, x, y, z)
        return self._next(collection)

    def spatial_interpolation_to_points(self, source_columns: Optional[List[str]] = None,
                                        target_columns: Optional[List[str]] = None,
                                        method: str = "inverse_distance", power: float = 2.0,
                                        *, result_prefix: str) -> Self:
        """補間先の列の座標位置での値を、補間元の列から行ごとに補間して "{result_prefix}{補間先の列名}" 列として追加します

        補間元と補間先がすべて z 座標を持つ場合は 3 次元、それ以外は x, y で補間します。

        Args:
            source_columns: 補間元の列（省略時は座標を持つ全列）
            target_columns: 補間先の列（省略時は補間元以外の座標を持つ列）
        """
        with_coords = self._require_coordinates()
        source_names = with_coords if source_columns is None else source_columns
        target_names = ([c for c in with_coords if c not in source_names]
                        if target_columns is None else self._check_columns(target_columns))
        if not target_names:
            raise ValueError("補間先の座標付き列がありません")

        target_coords = {name: self._c.get_column_coordinates(name) for name in target_names}
        target_coords = {n: c for n, c in target_coords.items() if c[0] is not None and c[1] is not None}
        use_z = (all(c[2] is not None for c in target_coords.values())
                 and all(self._c.get_column_coordinates(n)[2] is not None for n in source_names))
        names, points, values = self._sources(source_names, use_z)
        targets = np.array([c if use_z else c[:2] for c in target_coords.values()], dtype=float)
        results = interpolate(points, values, targets, method=method, power=power)

        collection = self._c
        unit = self._common_unit(names)
        for (target, (tx, ty, tz)), result in zip(target_coords.items(), results):
            name = f"{result_prefix}{target}"
            collection = collection.with_column(
                name, result, unit=unit, attrs={"interpolation": self._info(method, power, names, target_column=target)},
            ).with_coordinates(name, tx, ty, tz)
        return self._next(collection)

    def grid(self, *, x_range: Tuple[float, float], y_range: Tuple[float, float],
             grid_size: Tuple[int, int] = (10, 10), source_columns: Optional[List[str]] = None,
             row: int = 0, method: str = "inverse_distance", power: float = 2.0) -> GridResult:
        """指定した行の値を矩形グリッド上に補間します

        Args:
            x_range: X の範囲 (最小, 最大)
            y_range: Y の範囲 (最小, 最大)
            grid_size: グリッドの点数 (nx, ny)
            source_columns: 補間元の列（省略時は座標を持つ全列）
            row: 補間する行のインデックス（負の値は末尾から）

        Examples:
            >>> grid = coord.ops.grid(x_range=(0, 10), y_range=(0, 10), grid_size=(20, 20), row=-1)
            >>> grid.values.shape
            (20, 20)
        """
        n = len(self._c)
        if not -n <= row < n:
            raise IndexError(f"行インデックス {row} が範囲外です（行数 {n}）")
        names, points, values = self._sources(source_columns, use_z=False)
        nx, ny = grid_size
        xs, ys = np.linspace(*x_range, nx), np.linspace(*y_range, ny)
        gx, gy = np.meshgrid(xs, ys)
        grid = interpolate(points, values[:, row], np.column_stack([gx.ravel(), gy.ravel()]),
                           method=method, power=power).reshape(ny, nx)
        step = self._c.step[row]
        return GridResult(
            "grid", xs, ys, grid, unit=self._common_unit(names),
            metadata=self._info(method, power, names, row=row, step=step.item() if hasattr(step, "item") else step),
        )

    def nearest_neighbors(self, column: str, *, n_neighbors: int = 3) -> Dict[str, float]:
        """座標空間で column に近い列を、近い順に {列名: 距離} で返します（座標を持つ列が対象）"""
        if column not in self._c:
            raise KeyError(f"列 '{column}' が存在しません")
        valid = self._c.get_columns_with_coordinates()
        if len(valid) <= 1:
            raise ValueError("十分な数の座標データがありません。")
        if column not in valid:
            raise ValueError(f"カラム '{column}' には有効な座標が設定されていません。")

        candidates = [(name, self._c.distance(column, name)) for name in valid if name != column]
        neighbors = sorted((c for c in candidates if not np.isnan(c[1])), key=lambda c: c[1])[:n_neighbors]
        return {name: float(d) for name, d in neighbors}

    def clusters(self, *, n_clusters: int = 2, columns: Optional[List[str]] = None,

    def nearest_neighbors(self, column: str, *, n_neighbors: int = 3) -> Dict[str, float]:
    """座標空間で column に近い列を、近い順に {列名: 距離} で返します（座標を持つ列が対象）"""

    def spatial_interpolation_to_points(self, source_columns: Optional[List[str]] = None,
                                    target_columns: Optional[List[str]] = None,
                                    method: str = "inverse_distance", power: float = 2.0,
                                    *, result_prefix: str) -> Self:

    補間元と補間先がすべて z 座標を持つ場合は 3 次元、それ以外は x, y で補間します。

        source_columns: 補間元の列（省略時は座標を持つ全列）

    source_names = with_coords if source_columns is None else source_columns
    target_names = ([c for c in with_coords if c not in source_names]
                    if target_columns is None else self._check_columns(target_columns))
    if not target_names:
        raise ValueError("補間先の座標付き列がありません")

    target_coords = {name: self._c.get_column_coordinates(name) for name in target_names}
    target_coords = {n: c for n, c in target_coords.items() if c[0] is not None and c[1] is not None}

             and all(self._c.get_column_coordinates(n)[2] is not None for n in source_names))

            name, result, unit=unit, attrs={"interpolation": self._info(method, power, names, target_column=target)},
        ).with_coordinates(name, tx, ty, tz)
    return self._next(collection)

    def _rosette_vector_args(self, result_prefix: str, step_index: int) -> Dict[str, Any]:
        """calculate_rosette_strains の結果列 "{result_prefix}_e1" などから、指定ステップの主ひずみ・主方向と座標を取得します"""
        names = {key: f"{result_prefix}_{key}" for key in ("e1", "e2", "theta")}
        for name in names.values():
            if name not in self._collection:
                raise ValueError(f"計算済みカラム '{name}' が見つかりません。先に calculate_rosette_strains(result_prefix=\"{result_prefix}\") を実行してください。")
        n = len(self._collection)
        if not -n <= step_index < n:
            raise ValueError(f"Step index {step_index} is out of range.")

        x, y, _ = self._collection.get_column_coordinates(names["e1"])
        if x is None or y is None:
            raise ValueError(f"結果列 '{names['e1']}' に座標がありません（ロゼットのゲージ列に座標を設定してください）")

        values = {key: float(self._collection[name].values[step_index]) for key, name in names.items()}
        return {"x": x, "y": y, **values, "rosette_name": result_prefix, "step_index": step_index}

    def plot_rosette_vectors(self, result_prefix: str, step_index: int = 0, scale: float = 1.0,
                             ax: Optional[Axes] = None) -> Axes:
        """`calculate_rosette_strains(result_prefix=...)` の結果から、ロゼットの位置に主ひずみベクトルを描画します"""
        return strain_plot.plot_rosette_vectors(**self._rosette_vector_args(result_prefix, step_index),
                                                scale=scale, ax=ax)

    def _rosette_vectors(e1: float, e2: float, theta: float, scale: float = 1.0) -> Tuple[float, float, float, float]:
    """主ひずみ e1（theta 方向）と e2（theta + 90° 方向）の矢印の成分 (v1_x, v1_y, v2_x, v2_y)。theta は度"""
    rad = np.radians(theta)
    return (float(e1 * np.cos(rad) * scale), float(e1 * np.sin(rad) * scale),
            float(e2 * np.cos(rad + np.pi / 2) * scale), float(e2 * np.sin(rad + np.pi / 2) * scale))

    def plot_rosette_vectors(
    x: float,
    y: float,
    e1: float,
    e2: float,
    theta: float,
    rosette_name: str,
    step_index: int = 0,
    scale: float = 1.0,
    ax: Optional[Any] = None
) -> Any:
    """ロゼットの主ひずみベクトルをプロットします

    指定したステップにおける主ひずみ（最大・最小）の大きさと方向を、
    ロゼットの設置位置にベクトルとして描画します。

    Args:
        x: ロゼットのX座標
        y: ロゼットのY座標
        e1: 主ひずみ1 (最大主ひずみ)
        e2: 主ひずみ2 (最小主ひずみ)
        theta: 主ひずみ方向角度（度）
        rosette_name: ロゼット名 (タイトル表示用)
        step_index: ステップインデックス (タイトル表示用)
        scale: ベクトルの長さのスケール
        ax: MatplotlibのAxesオブジェクト（指定がない場合は新規作成）

    Returns:
        Axes: プロットされたAxesオブジェクト

    Examples:
        >>> from tascpy.domains.strain.draw import plot_rosette_vectors
        >>> ax = plot_rosette_vectors(x=0.0, y=0.0, e1=1000e-6, e2=-500e-6, theta=45.0, rosette_name="R1")
    """
    if ax is None:
        ax = new_axes()
        ax.set_title(f"Rosette: {rosette_name} (Step: {step_index})")

    v1_x, v1_y, v2_x, v2_y = _rosette_vectors(e1, e2, theta, scale)
    head = {"head_width": scale * 0.1, "head_length": scale * 0.1}
    ax.arrow(x, y, v1_x, v1_y, fc="r", ec="r", label=f"e1: {e1:.2e}", **head)
    ax.arrow(x, y, -v1_x, -v1_y, fc="r", ec="r", **head)
    ax.arrow(x, y, v2_x, v2_y, fc="b", ec="b", label=f"e2: {e2:.2e}", **head)
    ax.arrow(x, y, -v2_x, -v2_y, fc="b", ec="b", **head)
    ax.scatter([x], [y], color="k", marker="o", s=25)

    ax.set_aspect("equal")
    ax.grid(True)
    ax.set_xlabel("X")
    ax.set_ylabel("Y")
    return ax
